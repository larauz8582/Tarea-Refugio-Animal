class GestorSistema:
    _instancia = None

    def __new__(cls):
        if cls._instancia is None:
            cls._instancia = super().__new__(cls)
            cls._instancia.casos = []
            cls._instancia.usuarios = []
            print(" Sistema inicializado")
        return cls._instancia

    def registrar_caso(self, caso):
        self.casos.append(caso)
        print(f" Caso registrado: {caso.id}")

    def listar_casos(self):
        return self.casos
